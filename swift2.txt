program 3:  Login screen:


import SwiftUI

struct ContentView: View {
    var body: some View {
        VStack(spacing: 15) {
            Text("Login")
                .font(.largeTitle)
                .bold()

            TextField("Email", text: .constant(""))
                .textFieldStyle(.roundedBorder)

            SecureField("Password", text: .constant(""))
                .textFieldStyle(.roundedBorder)

            Button("Sign In") {
                print("Sign In Clicked")
            }
        }
        .padding()
    }
}

program 5:  counter UI



import SwiftUI

struct ContentView: View {
    @State private var count = 0

    var body: some View {
        VStack(spacing: 20) {
            Text("Counter App")
                .font(.largeTitle)

            Text("\(count)")
                .font(.system(size: 50))

            Button("Increase") {
                count += 1
            }
        }
    }
}
