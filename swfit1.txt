let apples = 17
let people = 5
print(apples / people) // 3 (whole number only)
print(apples % people) // 2 (what is left over)
print(Double(apples) / Double(people)) // 3.4 (decimal division)

output:

3
2
3.4

program 6: Button 

 
import SwiftUI

struct ContentView: View {
    var body: some View {
        VStack(spacing: 20) {
            Text("My First App")
                .font(.largeTitle)
                .bold()

            Button("Get Started") {
                print("Button clicked!")
            }
            .padding()
            .background(.blue)
            .foregroundStyle(.white)
            .clipShape(RoundedRectangle(cornerRadius: 12))
        }
        .padding()
    }
}

#Preview {
    ContentView()
}
