program 2: change text button



import SwiftUI

struct ContentView: View {
    @State private var message = "Welcome"

    var body: some View {
        VStack {
            Text(message)

            Button("Click Me") {
                message = "Hello, Swift!"
            }
        }
        .padding()
    }
}

program 5:  simple login screen


import SwiftUI

struct ContentView: View {
    var body: some View {
        VStack(spacing: 15) {
            Text("Login")
                .font(.title)

            TextField("Username", text: .constant(""))
                .textFieldStyle(.roundedBorder)

            SecureField("Password", text: .constant(""))
                .textFieldStyle(.roundedBorder)

            Button("Login") {
                print("Login clicked")
            }
            .buttonStyle(.borderedProminent)
        }
        .padding()
    }
}


program 6:  like button


import SwiftUI

struct ContentView: View {
    @State private var liked = false

    var body: some View {
        Button {
            liked.toggle()
        } label: {
            Label(
                liked ? "Liked" : "Like",
                systemImage: liked ? "heart.fill" : "heart"
            )
            .foregroundColor(.red)
        }
        .padding()
    }
}
