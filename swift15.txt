        }
    }
}


program3: arrange items on top of each other


import SwiftUI

struct ContentView: View {
    var body: some View {
        ZStack {
            Color.blue

            Text("Welcome")
                .font(.largeTitle)
                .foregroundColor(.white)
        }
    }
}


program4: vertical stack with spacing


import SwiftUI

struct ContentView: View {
    var body: some View {
        VStack(spacing: 30) {
            Text("Welcome")
            Text("Learn Swift")
            Text("Build Apps")
        }
    }
}


program5: horizontal stack with images


import SwiftUI

struct ContentView: View {
    var body: some View {
        HStack(spacing: 30) {
            Image(systemName: "house.fill")
                .foregroundColor(.blue)

            Image(systemName: "heart.fill")
                .foregroundColor(.red)

            Image(systemName: "person.fill")
                .foregroundColor(.green)
        }
        .font(.system(size: 35))
    }
}
