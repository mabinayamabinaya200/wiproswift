Program1:  Arithmetic Operation


let a = 20
let b = 10

print("Addition:", a + b)
print("Subtraction:", a - b)
print("Multiplication:", a * b)
print("Division:", a / b)
print("Remainder:", a % b)


program2:  Even or Odd 


let number = 15

if number % 2 == 0 {
    print("Even Number")
} else {
    print("Odd Number")
}


program3: Positive , Negative or Zero 


let number = -8

if number > 0 {
    print("Positive Number")
} else if number < 0 {
    print("Negative Number")
} else {
    print("Zero")
}


program4: Find the largest of two number


let a = 45
let b = 70

if a > b {
    print("Largest:", a)
} else {
    print("Largest:", b)
}


program5: simple calculator


let a = 25
let b = 5
let operation = "+"

switch operation {
case "+":
    print("Result:", a + b)
case "-":
    print("Result:", a - b)
case "*":
    print("Result:", a * b)
case "/":
    print("Result:", a / b)
default:
    print("Invalid Operation")
}
